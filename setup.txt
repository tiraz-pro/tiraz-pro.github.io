/* ═════════ أدوات ═════════ */
function folder_(parentId, name) {
  var q = "'" + parentId + "' in parents and name='" + name +
          "' and mimeType='application/vnd.google-apps.folder' and trashed=false";
  var r = Drive.Files.list({q: q, supportsAllDrives: true, includeItemsFromAllDrives: true,
                            corpora: 'drive', driveId: DRIVE_ID, fields: 'files(id,name)'});
  if (r.files && r.files.length) return r.files[0].id;
  var f = Drive.Files.create({name: name, mimeType: 'application/vnd.google-apps.folder',
                              parents: [parentId]}, null, {supportsAllDrives: true});
  return f.id;
}

function newSheet_(name, folderId) {
  var f = Drive.Files.create({name: name, mimeType: 'application/vnd.google-apps.spreadsheet',
                              parents: [folderId]}, null, {supportsAllDrives: true});
  return SpreadsheetApp.openById(f.id);
}

function tab_(ss, name, title, sub, headers, widths) {
  var sh = ss.getSheetByName(name) || ss.insertSheet(name);
  sh.setRightToLeft(true);
  if (title) {
    sh.getRange('A1').setValue(title).setFontSize(14).setFontWeight('bold').setFontColor(GOLD);
    if (sub) sh.getRange('A2').setValue(sub).setFontSize(9).setFontColor('#808080').setFontStyle('italic');
  }
  if (headers) {
    var r = sh.getRange(3, 1, 1, headers.length);
    r.setValues([headers]).setBackground(HEAD).setFontColor('#FFFFFF')
     .setFontWeight('bold').setHorizontalAlignment('center').setWrap(true);
    sh.setRowHeight(3, 34);
    sh.setFrozenRows(3);
  }
  if (widths) for (var i = 0; i < widths.length; i++) sh.setColumnWidth(i + 1, widths[i]);
  return sh;
}

function listRule_(vals) {
  return SpreadsheetApp.newDataValidation().requireValueInList(vals, true).setAllowInvalid(false).build();
}

/* ═════════ التشغيل ═════════ */
function setup() {
  var sys      = folder_(DRIVE_ID, '00 — النظام');
  var projects = folder_(DRIVE_ID, '01 — المشاريع');
  var guides   = folder_(DRIVE_ID, '02 — الأدلة');
  var camera   = folder_(DRIVE_ID, '03 — الكاميرات');

  var master = buildMaster_(sys);
  var tpl    = buildTemplate_(sys);

  var props = PropertiesService.getScriptProperties();
  props.setProperties({
    DRIVE_ID: DRIVE_ID, SYS_FOLDER: sys, PROJECTS_FOLDER: projects,
    GUIDES_FOLDER: guides, CAMERA_FOLDER: camera,
    MASTER_ID: master.getId(), TEMPLATE_ID: tpl.getId()
  });

  var out = [
    '✓ تم التأسيس',
    '',
    'مساحة العمل : https://drive.google.com/drive/folders/' + DRIVE_ID,
    'السجل الرئيسي: ' + master.getUrl(),
    'قالب المشروع : ' + tpl.getUrl(),
    '',
    'المجلدات: 00 — النظام · 01 — المشاريع · 02 — الأدلة · 03 — الكاميرات'
  ].join('\n');
  Logger.log(out);
  return out;
}
